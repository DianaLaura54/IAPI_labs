#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import time, threading, socket, struct
from xarm.wrapper import XArmAPI

# --- Minimal MQTT Client (QoS0) ---
class MiniMQTTClient:
    class _Msg:
        def __init__(self, topic, payload):
            self.topic = topic
            self.payload = payload

    def __init__(self, client_id=None, keepalive=30):
        self.client_id = client_id or f"mini-{int(time.time())}"
        self.keepalive = keepalive
        self.sock = None
        self.on_connect = None
        self.on_message = None
        self._stop = False
        self._last_io = time.time()
        self._lock = threading.Lock()

    def connect(self, host, port=1883):
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.sock.settimeout(5.0)
        self.sock.connect((host, port))
        self.sock.settimeout(0.2)
        self._send_connect()
        # Așteptare CONNACK
        end = time.time() + 5
        while time.time() < end:
            pkt = self._recv_packet()
            if pkt and pkt[0] == 2:
                if self.on_connect:
                    self.on_connect(self, None, {}, pkt[2][1])
                return pkt[2][1]
        return 1

    def subscribe(self, topic):
        pid = struct.pack("!H", 1)
        payload = pid + self._encode_str(topic) + b"\x00"
        self._send(0x82, payload)

    def loop_forever(self):
        while not self._stop:
            pkt = self._recv_packet()
            if pkt and pkt[0] == 3: # PUBLISH
                topic, msg_payload = self._parse_publish(pkt[2])
                if self.on_message:
                    self.on_message(self, None, self._Msg(topic, msg_payload))

    # --- Funcții interne pentru protocolul MQTT ---
    def _encode_str(self, s):
        b = s.encode(); return struct.pack("!H", len(b)) + b

    def _send(self, header, payload):
        rl = bytearray()
        n = len(payload)
        while True:
            d = n % 128; n //= 128
            if n > 0: d |= 0x80
            rl.append(d)
            if n == 0: break
        with self._lock:
            self.sock.sendall(bytes([header]) + rl + payload)

    def _send_connect(self):
        payload = self._encode_str("MQTT") + b"\x04\x02" + struct.pack("!H", self.keepalive) + self._encode_str(self.client_id)
        self._send(0x10, payload)

    def _recv_packet(self):
        try:
            b1 = self.sock.recv(1)
            if not b1: return None
            # Citire lungime variabilă și payload
            return (b1[0] >> 4, b1[0] & 0x0F, self.sock.recv(1024)) 
        except: return None

    def _parse_publish(self, payload):
        tlen = struct.unpack("!H", payload[0:2])[0]
        topic = payload[2:2+tlen].decode()
        msg = payload[2+tlen:].decode()
        return topic, msg

# --- Configurație Laborator ---
BROKER = "broker.emqx.io" 
TOPIC_GRIPPER = "xarm5/cmd/gripper" 
ROBOT_IP = "192.168.1.199" 

def on_message(c, u, m):
    payload = m.payload.strip()
    print(f"Comandă primită: {payload}")
    if payload == "1":
        arm.set_gripper_position(70, wait=True, speed=1000, auto_enable=True) 
        print("Gripper ÎNCHIS") 
    elif payload == "2":
        arm.set_gripper_position(600, wait=True, speed=1000, auto_enable=True) 
        print("Gripper DESCHIS") 

if __name__ == "__main__":
    # 1. Inițializare Robot
    arm = XArmAPI(ROBOT_IP, baud_checkset=False) 
    arm.clean_warn(); arm.clean_error() 
    arm.motion_enable(True); arm.set_mode(0); arm.set_state(0) 
    
    # 2. Configurare MQTT
    client = MiniMQTTClient()
    client.on_message = on_message
    client.on_connect = lambda c,u,f,rc: (print(f"Conectat la MQTT (rc={rc})"), c.subscribe(TOPIC_GRIPPER)) 
    client.connect(BROKER)
    
    # 3. Pornire buclă recepție
    threading.Thread(target=client.loop_forever, daemon=True).start() 
    print(f"Sistem gata. Trimite '1' (închide) sau '2' (deschide) pe topicul: {TOPIC_GRIPPER}")
    
    try:
        while True: time.sleep(1)
    except KeyboardInterrupt:
        print("Oprit.")
